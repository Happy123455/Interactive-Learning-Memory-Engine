function mapRange(value, inMin, inMax, outMin, outMax) {
    return (value - inMin) * (outMax - outMin) / (inMax - inMin) + outMin;
}

function speakConcept(text) {
    if (!narrationActive) return;
    window.speechSynthesis.cancel(); // Stop previous speech
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = 'en-US';
    utterance.pitch = mapRange(Math.random(), 0, 1, 0.8, 1.2); // Playful pitch
    utterance.rate = 1.0;
    window.speechSynthesis.speak(utterance);
}

function playSynthTone(oscillator, frequency, duration = 0.1) {
    if (!audioCtx) return;
    oscillator.frequency.setValueAtTime(frequency, audioCtx.currentTime);
    if (oscillator.gainNode) {
        oscillator.gainNode.cancelScheduledValues(audioCtx.currentTime);
        oscillator.gainNode.setValueAtTime(0.5, audioCtx.currentTime);
        oscillator.gainNode.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    }
}

function initAudio() {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    // Sun Oscillator (bright tone)
    sunOscillator = audioCtx.createOscillator();
    sunOscillator.type = 'sine';
    sunOscillator.gainNode = audioCtx.createGain();
    sunOscillator.connect(sunOscillator.gainNode).connect(audioCtx.destination);
    sunOscillator.gainNode.gain.setValueAtTime(0, audioCtx.currentTime); // Start silent
    sunOscillator.start();

    // Wind Oscillator (whooshing/noise-like tone)
    windOscillator = audioCtx.createOscillator();
    windOscillator.type = 'sawtooth';
    windOscillator.gainNode = audioCtx.createGain();
    windOscillator.connect(windOscillator.gainNode).connect(audioCtx.destination);
    windOscillator.gainNode.gain.setValueAtTime(0, audioCtx.currentTime); // Start silent
    windOscillator.start();
}

function drawGrid() {

    const gridSize = 20;
    for (let x = 0; x < canvas.width / dpi; x += gridSize) {

        ctx.stroke();

    for (let y = 0; y < canvas.height / dpi; y += gridSize) {
        ctx.beginPath();
        ctx.moveTo(0, y);

        ctx.stroke();
    }
}

function drawBuilding(orientationAngle) {
    ctx.save();
    ctx.translate(building.x + building.width / 2, building.y + building.height / 2);
    ctx.rotate(orientationAngle * Math.PI / 180); // Rotate building based on orientation
    ctx.fillStyle = building.color;
    ctx.strokeStyle = 'rgba(0, 255, 255, 0.5)'; // Neon outline
    ctx.lineWidth = 2;
    ctx.fillRect(-building.width / 2, -building.height / 2, building.width, building.height);
    ctx.strokeRect(-building.width / 2, -building.height / 2, building.width, building.height);

    // Dynamic "Sunlight Pour" animation on active facades
    const sunVector = {
        x: Math.cos((currentSunAngle + 90) * Math.PI / 180), // Approx sun direction
        y: Math.sin((currentSunAngle + 90) * Math.PI / 180)
    };

    const facadeColors = {
        north: { x: 0, y: -1 }, east: { x: 1, y: 0 }, south: { x: 0, y: 1 }, west: { x: -1, y: 0 }
    };

    // Determine which facade is facing the sun
    // For simplicity, just check direct orientation for now, later enhance with proper vector math
    const rotatedSunVector = { // sun's direction relative to building's rotated North
        x: sunVector.x * Math.cos(-orientationAngle * Math.PI / 180) - sunVector.y * Math.sin(-orientationAngle * Math.PI / 180),
        y: sunVector.x * Math.sin(-orientationAngle * Math.PI / 180) + sunVector.y * Math.cos(-orientationAngle * Math.PI / 180)
    };

    let illuminatedFacade = '';
    const tolerance = 0.5; // Angle tolerance for "facing"

    if (rotatedSunVector.y < -tolerance) illuminatedFacade = 'north';
    else if (rotatedSunVector.x > tolerance) illuminatedFacade = 'east';
    else if (rotatedSunVector.y > tolerance) illuminatedFacade = 'south';
    else if (rotatedSunVector.x < -tolerance) illuminatedFacade = 'west';

    // "Sunlight Pour" - glowing texture
    if (illuminatedFacade) {
        ctx.beginPath();
        ctx.fillStyle = `rgba(255, 165, 0, ${0.1 + Math.sin(Date.now() * 0.005) * 0.05})`; // Pulsing orange glow
        ctx.strokeStyle = 'orange';
        ctx.lineWidth = 3;
        // Drawing the specific facade
        switch (illuminatedFacade) {
            case 'north': ctx.fillRect(-building.width / 2, -building.height / 2, building.width, building.height * 0.1); break;
            case 'south': ctx.fillRect(-building.width / 2, building.height / 2 - building.height * 0.1, building.width, building.height * 0.1); break;
            case 'east': ctx.fillRect(building.width / 2 - building.width * 0.1, -building.height / 2, building.width * 0.1, building.height); break;
            case 'west': ctx.fillRect(-building.width / 2, -building.height / 2, building.width * 0.1, building.height); break;
        }
        ctx.stroke();
    }


    // Draw windows as simple rectangles (dynamic width based on windowRatio)
    ctx.fillStyle = 'rgba(100, 200, 255, 0.5)';
    const windowWidth = building.width * 0.2;
    const windowHeight = building.height * building.windowRatio;

    // Front (South if orientation is 0) window
    ctx.fillRect(-windowWidth / 2, building.height / 2 - windowHeight - 5, windowWidth, windowHeight);
    // Add more windows if desired for other facades, based on orientation
    // For simplicity, just one window on "south" for now, relative to rotated building.

    ctx.restore();
}

function drawSunPath(angle) {
    const centerX = canvas.width / (2 * dpi);
    const centerY = canvas.height / (2 * dpi);
    const radius = Math.min(centerX, centerY) * 0.6;

    ctx.save();
    ctx.translate(centerX, centerY);

    // Sun path arc
    ctx.beginPath();
    ctx.arc(0, 0, radius, Math.PI, 0); // Semicircle from West to East
    ctx.strokeStyle = 'rgba(255, 255, 0, 0.3)';
    ctx.lineWidth = 2;
    ctx.stroke();

    // Sun position
    const sunX = radius * Math.cos(angle * Math.PI / 180);
    const sunY = -radius * Math.sin(angle * Math.PI / 180); // Y-axis inverted for canvas
    ctx.beginPath();
    ctx.arc(sunX, sunY, 10, 0, Math.PI * 2);
    ctx.fillStyle = 'yellow';
    ctx.shadowBlur = 15;
    ctx.shadowColor = 'yellow';
    ctx.fill();
    ctx.shadowBlur = 0; // Reset shadow

    // Sun rays
    for (let i = 0; i < 8; i++) {
        ctx.beginPath();
        ctx.moveTo(sunX + Math.cos(i * Math.PI / 4) * 12, sunY + Math.sin(i * Math.PI / 4) * 12);
        ctx.lineTo(sunX + Math.cos(i * Math.PI / 4) * 20, sunY + Math.sin(i * Math.PI / 4) * 20);
        ctx.strokeStyle = 'rgba(255, 255, 0, 0.7)';
        ctx.lineWidth = 1;
        ctx.stroke();
    }

    // Dynamic Shadows - (simplified, cast from building)
    ctx.save();
    ctx.translate(-building.width / 2, -building.height / 2); // Translate to building's top-left corner
    const shadowOffsetX = Math.cos((angle + 90) * Math.PI / 180) * 20; // Angle 0 = sun from west
    const shadowOffsetY = Math.sin((angle + 90) * Math.PI / 180) * 20;

    // Draw a basic shadow polygon behind the building
    ctx.fillStyle = 'rgba(0, 0, 0, 0.4)';
    ctx.beginPath();
    ctx.moveTo(building.x + building.width, building.y + building.height);
    ctx.lineTo(building.x + building.width + shadowOffsetX, building.y + building.height + shadowOffsetY);
    ctx.lineTo(building.x + shadowOffsetX, building.y + building.height + shadowOffsetY);
    ctx.lineTo(building.x, building.y + building.height);
    ctx.fill();
    ctx.restore();

    ctx.restore();
}

function updateWindParticles() {
    const windAngleRad = building.orientation * Math.PI / 180; // Wind direction opposite to building's "front"
    const windDirectionX = Math.cos(windAngleRad);
    const windDirectionY = Math.sin(windAngleRad);

    for (let i = 0; i < windParticles.length; i++) {
        const p = windParticles[i];
        p.x += windDirectionX * p.speed;
        p.y += windDirectionY * p.speed;
        p.life--;

        if (p.life <= 0 || p.x < 0 || p.x > canvas.width / dpi || p.y < 0 || p.y > canvas.height / dpi) {
            // Reset particle if it goes off screen or dies
            p.x = Math.random() * canvas.width / dpi;
            p.y = Math.random() * canvas.height / dpi;
            p.life = Math.random() * 50 + 50;
            p.speed = Math.random() * currentWindSpeed * 0.5 + 1;
        }
    }
}

function drawWindFlow() {
    ctx.fillStyle = `rgba(0, 255, 255, ${mapRange(currentWindSpeed, 0, 10, 0.05, 0.2)})`; // Wind current glow
    windParticles.forEach(p => {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
    });

    // Wind direction arrow (relative to building for simplicity)
    ctx.save();
    ctx.translate(building.x + building.width / 2, building.y + building.height / 2);
    ctx.rotate((building.orientation + 90) * Math.PI / 180); // Rotate based on building orientation for visual clarity
    ctx.strokeStyle = `rgba(0, 255, 0, ${mapRange(currentWindSpeed, 0, 10, 0.3, 1)})`;
    ctx.fillStyle = `rgba(0, 255, 0, ${mapRange(currentWindSpeed, 0, 10, 0.3, 1)})`;
    ctx.lineWidth = 3;

    ctx.beginPath();
    ctx.moveTo(-50, 0);
    ctx.lineTo(-10, 0);
    ctx.stroke();

    ctx.beginPath(); // Arrow head
    ctx.moveTo(-10, 0);
    ctx.lineTo(-20, -7);
    ctx.lineTo(-20, 7);
    ctx.closePath();
    ctx.fill();

    ctx.restore();
}

function drawComfortMeter() {
    const comfort = calculateOverallComfort(); // Get comfort value
    const centerX = canvas.width / (2 * dpi) + building.width / 2 + 50;
    const centerY = canvas.height / (2 * dpi) - 50;

    let emoji = '😐';
    let color = 'gray';
    if (comfort > 70) { emoji = '😁'; color = var_to_rgb('--neon-green'); }
    else if (comfort > 40) { emoji = '😊'; color = var_to_rgb('--neon-cyan'); }
    else if (comfort < 20) { emoji = '😥'; color = 'red'; }

    ctx.font = '48px sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.shadowBlur = 10;
    ctx.shadowColor = color;
    ctx.fillText(emoji, centerX, centerY);
    ctx.shadowBlur = 0; // Reset shadow

    // Add a bounce animation class to the element in the DOM for extra fun
    overallComfortScoreDisplay.textContent = emoji;
    overallComfortScoreDisplay.classList.toggle('bouncy-element', comfort > 70 || comfort < 20);
}

function var_to_rgb(cssVar) {
    const style = getComputedStyle(document.body);
    const value = style.getPropertyValue(cssVar);
    // Simple regex to extract RGB values if available
    const match = value.match(/(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/);
    if (match) return `rgb(${match[1]},${match[2]},${match[3]})`;
    return value; // Return as is if not RGB
}

// --- SIMULATION LOGIC ---
function calculateSolarGain() {
    // Simplified: more sun exposure at acute angles, less with good shading.
    // Assuming building's "front" is sensitive to sun.
    const relativeSunAngle = Math.abs((currentSunAngle - building.orientation + 360) % 360); // Angle relative to building's main axis
    // If sun is roughly perpendicular to a large facade (e.g., within 45 deg)
    let directGain = 0;
    if (relativeSunAngle < 45 || relativeSunAngle > 315 || (relativeSunAngle > 135 && relativeSunAngle < 225)) {
        directGain = mapRange(Math.cos(relativeSunAngle * Math.PI / 180), -1, 1, 0, 100);
    }
    // Reduce if there's an overhang (simulated for simplicity)
    if (building.overhangDepth > 0 && (relativeSunAngle > 45 && relativeSunAngle < 135)) { // Sun from the "north" if front is south
        directGain *= 0.7; // Simple reduction
    }
    return directGain;
}

function calculateWindVentilation() {
    // Simplified: better ventilation if wind is perpendicular to building, creating cross-ventilation.
    const relativeWindAngle = Math.abs((building.orientation - 90 + 360) % 360); // Wind hits a side (e.g., East/West)
    let ventilation = 0;
    if (relativeWindAngle < 45 || relativeWindAngle > 315 || (relativeWindAngle > 135 && relativeWindAngle < 225)) {
        ventilation = mapRange(currentWindSpeed, 0, 10, 0, 100) * (0.5 + Math.abs(Math.cos(relativeWindAngle * Math.PI / 180)) * 0.5);
    }
    return ventilation;
}

function calculateShadingEffect() {
    // Simplified: Better shading if sun is blocked effectively by orientation or overhangs
    const relativeSunAngle = Math.abs((currentSunAngle - building.orientation + 360) % 360);
    let shading = 0;
    // Good shading if sun is high (above 60 deg) and building is oriented North-South (or has overhangs)
    if (currentSunAngle > 60 && (relativeSunAngle < 45 || relativeSunAngle > 315)) {
        shading = 50; // Base shading from orientation
    }
    if (building.overhangDepth > 0 && currentSunAngle > 30) {
        shading += mapRange(currentSunAngle, 30, 90, 0, 50); // Additional shading from overhang
    }
    return shading;
}


function calculateOverallComfort() {
    const solar = calculateSolarGain();
    const ventilation = calculateWindVentilation();
    const shading = calculateShadingEffect();

    let comfort = 0;
    if (currentClimateZone === 'hot-dry') {
        // In hot-dry, minimize solar gain, maximize shading. Ventilation is good if cool enough.
        comfort = (100 - solar) * 0.4 + shading * 0.4 + ventilation * 0.2;
    } else if (currentClimateZone === 'warm-humid') {
        // In warm-humid, maximize ventilation, minimize solar gain. Shading helps.
        comfort = (100 - solar) * 0.3 + ventilation * 0.5 + shading * 0.2;
    } else {
        // Default / temperate zone
        comfort = (100 - solar) * 0.3 + ventilation * 0.3 + shading * 0.4;
    }

    return Math.min(100, Math.max(0, comfort)); // Clamp between 0-100
}


// --- MAIN SIMULATION LOOP ---
function drawSimulation() {
    ctx.clearRect(0, 0, canvas.width / dpi, canvas.height / dpi);
    drawGrid();

    drawWindFlow();
    drawComfortMeter();

    updateMetricsDisplay();
    updateBarCharts();

    if (simulationRunning) {
        animationFrameId = requestAnimationFrame(drawSimulation);
    }
}

    updateWindParticles();
}

function loop() {
    updateSimulation();
    drawSimulation();
}

// Building Rotation (Hand/Mouse Gesture)
canvas.addEventListener('mousedown', (e) => {
    const rect = canvas.getBoundingClientRect();
    mouse.x = (e.clientX - rect.left) / dpi;
    mouse.y = (e.clientY - rect.top) / dpi;

    const buildingCenterX = building.x + building.width / 2;
    const buildingCenterY = building.y + building.height / 2;

    // Check if clicking on the building for rotation
    if (mouse.x > building.x && mouse.x < building.x + building.width &&
        mouse.y > building.y && mouse.y < building.y + building.height) {
        isDraggingBuilding = true;
        startDragAngle = building.orientation;
        startMouseAngle = Math.atan2(mouse.y - buildingCenterY, mouse.x - buildingCenterX) * 180 / Math.PI;
        canvas.style.cursor = 'grabbing';
        speakConcept("Ooh, spinning the building! Watch it find its best side!");
    }
});

canvas.addEventListener('mousemove', (e) => {
    const rect = canvas.getBoundingClientRect();
    mouse.x = (e.clientX - rect.left) / dpi;
    mouse.y = (e.clientY - rect.top) / dpi;

    if (isDraggingBuilding) {
        const buildingCenterX = building.x + building.width / 2;
        const buildingCenterY = building.y + building.height / 2;
        const currentMouseAngle = Math.atan2(mouse.y - buildingCenterY, mouse.x - buildingCenterX) * 180 / Math.PI;
        let deltaAngle = currentMouseAngle - startMouseAngle;
        building.orientation = (startDragAngle + deltaAngle + 360) % 360; // Keep between 0-360
        // Small chime sound on rotation
        playSynthTone(sunOscillator, 220, 0.05); // Subtle click
        drawSimulation(); // Update faster during drag
    }

    // Hand/Mouse Motion Gesture Detection (Sweep to change orientation if not dragging building)
    if (!isDraggingBuilding && e.buttons === 0) { // Only if no button is pressed
        if (prevMouseX !== 0) {
            const sweepThreshold = 5; // Pixels to detect a sweep
            if (Math.abs(mouse.x - prevMouseX) > sweepThreshold) {
                const deltaX = mouse.x - prevMouseX;
                building.orientation = (building.orientation + deltaX * 0.5 + 360) % 360; // Adjust sensitivity
                playSynthTone(windOscillator, 150 + Math.abs(deltaX) * 5, 0.05);
                speakConcept("Whoosh! Sweeping your hand changes the building's direction!");
            }
        }
        prevMouseX = mouse.x;
    }
});

canvas.addEventListener('mouseup', () => {
    isDraggingBuilding = false;
    canvas.style.cursor = 'grab';
    prevMouseX = 0; // Reset prevMouseX after interaction ends
    speakConcept(`The building is now facing ${building.orientation.toFixed(0)} degrees!`);
});

canvas.addEventListener('mouseleave', () => {
    isDraggingBuilding = false;
    canvas.style.cursor = 'grab';
    prevMouseX = 0;
});
